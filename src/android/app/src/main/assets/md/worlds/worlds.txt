#   [FAMILY] Name

#   story  = one line
#   tempo  = BPM                        (optional)
#   macros = DENSITY MOTION COLOUR SPACE  in %   (optional)
#   engine = NAME TRACK VALUE; ...      MUTE, LEVEL, SEND, CUTOFF, RESONANCE, DECAY, TONE, HARM, PAN, ATTACK,
#                                       BREATHE, PROB (with a track 1-6, values in %, MUTE 0/1);
#                                       SWING, DELAY_TIME (ms), DELAY_FB, REVERB_TIME, FX_LEVEL, GRAIN_SIZE,
#                                       GRAIN_POS, FEEDBACK, GRAIN_PITCH (semitones), REVERSE, FREEZE, ROOT,
#                                       TRANSPOSE (without a track)
#   lab    = MODULE.NAME VALUE; ...     as on the Sound Lab's pages, _ for spaces (LOGIC.RANGE_(OCTAVES) 2)
#   sleep  = MINUTES                    (optional) the SLEEP TIMER: 15, 30, 60 or 90, set when none runs yet
# Entering a world sets the Sound Lab back to neutral first (the chosen sample and voice files stay).
# LOGIC.RULE: 0 TINTINNABULI 1 PHASING 2 FIBONACCI 3 AUTOMATON 4 MARKOV 5 L-SYSTEM 6 HARMONIC
